function exists(rel) {
  return fs.existsSync(path.join(ROOT, rel));
}
function read(rel) {
  const p = path.join(ROOT, rel);
  return fs.existsSync(p) ? fs.readFileSync(p, 'utf8') : '';
}
function write(rel, content) {
  const p = path.join(ROOT, rel);
  fs.mkdirSync(path.dirname(p), { recursive: true });
  const prev = exists(rel) ? fs.readFileSync(p, 'utf8') : null;
  if (prev === content) return false;
  fs.writeFileSync(p, content.endsWith('\n') ? content : content + '\n');
  console.log('Wrote', rel);
  return true;
}
function has(rel, needle) {
  return read(rel).includes(needle);
}
function countImgs(dir) {
  const abs = path.join(ROOT, dir);
  if (!fs.existsSync(abs)) return 0;
  return fs.readdirSync(abs).filter((f) => /\.(png|jpe?g|webp|gif)$/i.test(f)).length;
}
function findReadmes(dir, acc) {
  acc = acc || [];
  for (const name of fs.readdirSync(dir)) {
    if (name === '.git' || name === 'node_modules') continue;
    const full = path.join(dir, name);
    const st = fs.statSync(full);
    if (st.isDirectory()) findReadmes(full, acc);
    else if (name === 'README.md') acc.push(path.relative(ROOT, full).replace(/\\/g, '/'));
  }
  return acc;
}
